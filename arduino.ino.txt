int led = 13;
int button = 2;

void setup()
{
  pinMode(led, OUTPUT);
  pinMode(button, INPUT);
}

void loop()
{
  int state = digitalRead(button);

  if(state == HIGH)      // Button pressed
  {
    digitalWrite(led, HIGH); // LED ON
    delay(5000);             // Wait for 5 seconds
    digitalWrite(led, LOW);  // LED OFF
  }
}
